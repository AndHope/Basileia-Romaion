	# Using this in a different strategy is inclusive, only one needs to return true
	# root = country, scope:target_state = the state being considered
	will_incorporate_state = {

			limit = {
				country_is_colonial_or_company = yes
			}
			scope:target_state = { ai_colony_will_incorporate_state = yes }
		}
		trigger_else_if = { # Default behaviour
			limit = {
				scope:target_state = { ai_will_incorporate_state = yes }
			}
			always = yes
		}
		trigger_else = { # Content
			OR = {
				AND = {
					has_journal_entry = je_the_algerian_departements
					scope:target_state = {
						OR = {
							state_region = s:STATE_ORAN
							state_region = s:STATE_ALGIERS
							state_region = s:STATE_CONSTANTINE
						}
					}
				}
				AND = {
					has_journal_entry = je_french_natural_borders

						OR = {
							state_region = s:STATE_WALLONIA
							state_region = s:STATE_FLANDERS
							state_region = s:STATE_RHINELAND
							state_region = s:STATE_NORTH_RHINE
						}
					}
				}
				AND = {
					has_journal_entry = je_matter_of_hungary
					scope:target_state = {
						state_region = { is_homeland = cu:hungarian }
					}
				}

					has_journal_entry = je_la_trinitaria

						is_in_geographic_region = geographic_region_korea
					}
				}
				AND = {
					has_journal_entry = je_tp_pool_develop_frontier
					scope:target_state = {
						OR = {
							has_variable = rus_develop_frontier_unincorporated_state

								root = { has_variable = tp_siberia_frontier_target_gdp }
								is_in_geographic_region = geographic_region_siberia

								root = { has_variable = tp_central_asia_frontier_target_gdp }

	# Using this value in other strategies will function additively

	# scope:law = law type being enacted
	# scope:interest_group = IG to negotiate with
	# scope:amenability = IG amenability to negotiation divided by MAX_AMENABILITY (ie, if MAX_AMENABILITY is 100 and IG amenability is 50, this value will be 0.5)
	# scope:advance_chance = Chance to advance law enactment
	# scope:stall_chance = Chance for law enactment to stall
	negotiation_chance = {
		value = 0

		if = {
			limit = {
				scope:interest_group ?= {
					amenability_level != 4
				}

				limit = {
					ai_has_reasons_to_not_oppose_law = yes
				}
				multiply = 2
			}

			subtract = {
				value = scope:advance_chance
				multiply = 2
			}
		}

		max = 1
	}

	# root = country
	# scope:target_state = state
	cossack_host_scores = {
		establish_cossack_host = {
			value = 0

			if = {
				limit = {
					scope:target_state = {
						is_incorporated = no
						OR = {
							region = sr:region_central_asia
							region = sr:region_russia
							region = sr:region_siberia
							region = sr:region_greater_persia
							region = sr:region_north_china
							region = sr:region_northeast_asia
						}
					}
				}
				add = 15
			}

			if = {
				limit = {
					scope:target_state = {
						any_neighbouring_state = {
							owner ?= {
								OR = {
									is_country_type = unrecognized
									is_country_type = decentralized
								}
							}
						}
					}
				}
				add = 10

				if = {
					limit = {
						scope:target_state = {
							is_split_state = yes
						}
					}
					add = 10
				}
			}

			if = {
				limit = {
					scope:target_state = {
						is_under_colonization = yes
					}
				}
				add = 75
			}

			if = {
				limit = {
					scope:target_state = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.9
						}
					}
				}
				multiply = 0
			}
			else_if = {
				limit = {
					scope:target_state = {
						primary_cultures_percent_state = {
							target_country = owner
							value >= 0.8
						}
					}
				}
				multiply = 0.5
			}

			if = {
				limit = {
					OR = {
						has_law_or_variant = law_type:law_council_republic
						has_law_or_variant = law_type:law_state_atheism
						has_law_or_variant = law_type:law_command_economy
						has_law_or_variant = law_type:law_multicultural
					}
				}
				multiply = 0
			}

			if = {
				limit = {
					scope:target_state = {
						has_recently_removed_cossack_host = yes
					}
				}
				multiply = 0
			}
		}

		disband_cossack_host = {
			value = 0

			if = {
				limit = {
					scope:target_state = {
						is_incorporated = yes
					}
				}
				add = 10
			}

			if = { # Russia should be very unlikely to remove Cossacks
				limit = {
					NOT = { has_law_or_variant = law_type:law_council_republic }
					any_primary_culture = {
						has_discrimination_trait = heritage_east_slavic
					}
				}
				multiply = 0.5

				if = {
					limit = {
						OR = {
							has_law = law_type:law_monarchy
							has_law = law_type:law_internal_colonization
						}
					}
					multiply = 0.25
				}
			}
			else = { # Everyone else should remove Cossacks
				add = 1000
			}
		}
	}

	# Determines which strait settings the AI will use, highest score
	# Using this value in other strategies will function additively
	# root = country
	# scope:strait = the strait
	# scope:first_state = the first state controlling the strait
	# scope:second_state = the second state controlling the strait
	# scope:is_sole_controller = yes/no depending on whether country controls both sides of the strait
	strait_scores = {
		status_scores = {
			open = {
				value = 1 # Options scoring exactly 0 are skipped, so keep opening up available
			}

			no_military = {
				value = 10

				subtract = { # Fallout with the countries this would hurt
					value = {
						scope:strait = {
							add = "military_fallout_for(root)"
						}
					}
					multiply = ai_strait_fallout_weight
				}
				min = 0
			}

			closed = {
				value = 0

				if = { # Close canals to enemy trade
					limit = {
						scope:strait = {
							OR = {
								type = strait_type:canal_panama
								type = strait_type:canal_suez
							}
						}
					}

					add = 100
				}

				subtract = { # Fallout with the countries this would hurt
					value = {
						scope:strait = {
							add = "fallout_for(root)"
						}
					}
					multiply = ai_strait_fallout_weight
				}
				min = 0
			}
		}

		permissions_scores = {
			all = {
				value = 0
			}

			no_enemies = {
				value = 10
			}

			only_allies = {
				value = 0

				if = { # Wants to keep competitor navies out
					limit = {
						OR = {
							has_strategy = ai_strategy_colonial_expansion
							has_strategy = ai_strategy_territorial_expansion
						}
					}

					add = 20
				}

				subtract = { # Fallout with the non-allies this would shut out
					value = {
						scope:strait = {
							add = "only_allies_fallout_for(root)"
						}
					}
					multiply = ai_strait_fallout_weight
				}
				min = 0

				if = { # Don't close off gibraltar
					limit = {
						scope:strait = {
							type = strait_type:strait_gibraltar
						}
					}

					multiply = 0
				}
			}

			only_controller = {
				value = 0
			}
		}

		toll_rate_scores = {
			no_tolls = {
				value = 1 # Options scoring exactly 0 are skipped, so keep dropping tolls available
			}

			low_tolls = {
				value = 10

				if = {
					limit = {
						scope:is_sole_controller = yes
						OR = {
							has_strategy = ai_strategy_maintain_power_balance
							has_strategy = ai_strategy_anti_imperialism
						}
					}

					add = 20
				}

				add = { # The income the tolls would bring
					value = scope:toll_income_per_percent
					multiply = define:NEconomy|TOLL_RATE_LOW_TOLLS
					multiply = 100 # The toll rate defines are fractions, the income is per percent
					multiply = ai_strait_toll_income_weight
				}

				subtract = { # Fallout with the countries this would hurt
					value = {
						scope:strait = {
							add = "trade_fallout_for(root)"
						}
					}
					multiply = ai_strait_fallout_weight
					multiply = define:NEconomy|TOLL_RATE_LOW_TOLLS # The share of their crossing trade the tolls would take
					divide = 2
					if = {
						limit = {
							scope:is_sole_controller = yes
						}
						multiply = define:NEconomy|TOLL_RATE_MULTIPLIER_FULL_CONTROL
					}
				}
				min = 0
			}

			medium_tolls = {
				value = 0

				if = { # "Greed is good"
					limit = {
						OR = {
							has_strategy = ai_strategy_economic_imperialism
							has_strategy = ai_strategy_colonial_expansion
						}
					}

					add = 20
				}

				if = { # Can charge higher tolls if we control both sides
					limit = {
						scope:is_sole_controller = yes
					}

					add = 20
				}

				if = { # Don't extort in gibraltar
					limit = {
						scope:strait = {
							type = strait_type:strait_gibraltar
						}
					}

					multiply = 0.5
				}
				else_if = { # Make some money on your canals
					limit = {
						scope:strait = {
							OR = {
								type = strait_type:canal_panama
								type = strait_type:canal_suez
							}
						}
					}

					add = 100
				}

				add = { # The income the tolls would bring
					value = scope:toll_income_per_percent
					multiply = define:NEconomy|TOLL_RATE_MEDIUM_TOLLS
					multiply = 100 # The toll rate defines are fractions, the income is per percent
					multiply = ai_strait_toll_income_weight
				}

				subtract = { # Fallout with the countries this would hurt
					value = {
						scope:strait = {
							add = "trade_fallout_for(root)"
						}
					}
					multiply = ai_strait_fallout_weight
					multiply = define:NEconomy|TOLL_RATE_MEDIUM_TOLLS # The share of their crossing trade the tolls would take
					if = {
						limit = {
							scope:is_sole_controller = yes
						}
						multiply = define:NEconomy|TOLL_RATE_MULTIPLIER_FULL_CONTROL
					}
				}
				min = 0
			}

			high_tolls = {
				value = 0
			}
		}
	}

	# A fleet composition describes a role a fleet can take (e.g. main_battle_fleet, raid_fleet):
	# what kind of ships it should be made of and what missions it should pursue. The AI will try
	# to match each of the country's existing fleets to the role they best fit.
	# All numeric fields accept script values, so weights can be tuned from country scope.
	#
	# ai_ship_category_weights:
	#   Per AI ship category desire for this role, see common/ai_ship_categories. Across all
	#   available roles, weights are summed per category, and each role's share becomes its target
	#   ratio of the country's total ships in that category. Existing fleets are scored against
	#   every role's target ratios and assigned to the role they fit best, so these weights drive
	#   both "how should the country's ships be split across roles" and "which fleet plays which
	#   role".
	#   A category left at 0 here is never allocated to this role, so a category that only one role
	#   wants is kept out of every other fleet.
	#
	# mission_type_weights:
	#   Multiplier applied when a fleet in this role picks which mission to pursue. 1.0 is
	#   neutral, higher biases the fleet toward that mission, 0.0 disables it for this role.
	#   IMPORTANT: if the block is present, any mission type not listed is treated as 0, so the
	#   list gates which missions this role will take. Omit the whole block to leave every
	#   mission neutral.
	#
	# min_ships_wanted:
	#   Soft floor on fleet size used during role-fit scoring. Fleets smaller than this take a
	#   penalty when matched to this role; it does not disqualify them and is not a hard cap.
	#
	# num_fleets_wanted:
	#   How many fleets should take this role. Omitted means one. Evaluated in country scope, so
	#   it can scale with navy size, and a value of 0 drops the role for that country.
	#
	# potential:
	#   Country-scoped trigger. If it fails the role is dropped entirely, so no fleet will be
	#   assigned to it.
	#
	# replace_key:
	#   Optional key of another composition that this one should replace when both are collected
	#   from the country's active strategies. Use it in a country-specific strategy to override a
	#   default role (e.g. a custom raid_fleet with replace_key = raid_fleet swaps out the one
	#   defined here). Without a replace_key the composition is added alongside the existing set.
