INJECT:law_state_religion = {
	modifier = {
		country_clergymen_pol_str_mult = 0.2 #additional 0.2 for 0.5 total
		country_state_religion_wages_mult = -0.05 #subtraction from 0.1 for 0.05 total
		country_non_state_religion_wages_mult = -0.05 #subtraction from -0.1 for -0.15 total
	}
}

REPLACE_OR_CREATE:law_caesaropapism = {
	group = lawgroup_church_and_state
	parent = law_state_religion # Inherits ideological stances from parent

	icon = "gfx/interface/icons/law_icons/caesaropapism.dds"

	progressiveness = -50

	is_visible = {
		OR = {
			c:RUS ?= this
			AND = {
				exists = c:RUS
				is_subject_of = c:RUS
				country_is_subject_type_valid_for_caesaropapism = yes
			}
			#BR added
			c:BYZ ?= this
			AND = {
				exists = c:BYZ
				is_subject_of = c:BYZ
				country_is_subject_type_valid_for_caesaropapism = yes
			}
		}
		NOT = {
			has_law_or_variant = law_type:law_theocracy
		}
	}

	on_activate = {
		if = {
			limit = {
				has_modifier = antireligious_policies
			}
			remove_modifier = antireligious_policies
		}
		recalculate_pop_ig_support = yes
	}

	modifier = {
		interest_group_ig_devout_pol_str_mult = 0.15
		country_bureaucrats_pol_str_mult = 0.5
		country_authority_add = 200
		country_state_religion_wages_mult = 0.05 #br change from 0.1
		country_non_state_religion_wages_mult = -0.15 #br change from 0.1

		country_acceptance_state_religion_add = 25
		country_acceptance_shared_religious_trait_add = 0
		country_acceptance_shared_religious_trait_group_add = -10
		country_acceptance_no_shared_religious_trait_add = -25

		country_allow_conversion_violent_hostility_bool = yes
		country_allow_conversion_cultural_erasure_bool = yes
		country_allow_conversion_open_prejudice_bool = yes
		country_allow_conversion_second_rate_citizen_bool = yes
		country_allow_conversion_full_acceptance_bool = yes
	}

	can_impose = {
		OR = {
			AND = {
				is_in_same_power_bloc = scope:target_country
				AND = {
					modifier:country_can_impose_same_lawgroup_church_and_state_in_power_bloc_bool = yes
					has_law = scope:law
				}
			}
			can_impose_law_default = yes
		}
	}

	ai_will_do = {
		exists = ruler
		ruler = {
			OR = {
				has_ideology = ideology:ideology_theocrat
				has_ideology = ideology:ideology_traditionalist
			}
		}
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0

		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}

	ai_impose_chance = {
		value = 0

		if = {
			limit = {
				has_law = law_type:law_caesaropapism
				power_bloc ?= { has_identity = identity:identity_religious }
				religion = scope:target_country.religion
			}

			add = base_impose_law_weight

			# Very important for religious blocs
			if = {
				limit = { scope:target_country = { has_law_or_variant = law_type:law_freedom_of_conscience } }
				multiply = 2
			}
			else = {
				multiply = 5
			}
		}
	}
}

INJECT:law_freedom_of_conscience = {
	unlocking_technologies = {
		br_tech_skepticism
	}
}
