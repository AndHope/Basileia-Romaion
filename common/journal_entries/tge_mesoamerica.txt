je_tge_golden_eagle = {
	group = je_group_historical_content
	icon = "gfx/interface/icons/event_icons/event_aztec.dds"
	weight = 100
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	is_shown_in_lobby = {
		c:NAH ?= this
	}
	
	scripted_progress_bar = tge_golden_eagle_progressbar
	
	immediate = {
		set_variable = {
			name = tge_golden_eagle_var
			value = 100
		}
		set_variable = {
			name = tge_golden_eagle_progress_var
			value = 0
		}
	}

	modifiers_while_active = {
		modifier_tge_golden_aspirations
	}

    # Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_tge_golden_eagle_completed_desc
		color = "gold_positive"
		trigger = {
			custom_tooltip = {
				text = je_tge_golden_eagle_complete_1
				var:tge_golden_eagle_var >= 500
			}
			custom_tooltip = {
				text = je_tge_golden_eagle_complete_2
				owns_entire_state_region = STATE_MEXICO
				capital ?= {
					state_region = s:STATE_MEXICO
				}
			}
			calc_true_if = {
				amount >= 10
				owns_entire_state_region = STATE_CHIHUAHUA
				owns_entire_state_region = STATE_DURANGO
				owns_entire_state_region = STATE_GUERRERO
				owns_entire_state_region = STATE_JALISCO
				owns_entire_state_region = STATE_OAXACA
				owns_entire_state_region = STATE_RIO_GRANDE
				owns_entire_state_region = STATE_SINALOA
				owns_entire_state_region = STATE_SONORA
				owns_entire_state_region = STATE_VERACRUZ
				owns_entire_state_region = STATE_ZACATECAS
			}
			has_law = law_type:law_monarchy
			has_law = law_type:law_autocracy
			has_law = law_type:law_state_religion
			country_rank >= rank_value:major_power
		}
		effect = {
			trigger_event = {
				id = tge_mesoamerica.109
			}
		}
	}

    # Fail
	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		conclusion_desc = je_tge_golden_eagle_failed_desc
		color = "blue_negative"
		trigger = {
			OR = {
				custom_tooltip = {
					text = je_tge_golden_eagle_fail
					var:tge_golden_eagle_var <= 0
				}
				NOT = {
					country_has_state_religion = rel:br_tge_religion_teotl
				}
				is_subject = yes
			}
		}
		effect = {
			trigger_event = {
				id = tge_mesoamerica.108
			}
		}
	}

	on_weekly_pulse = {
		effect = {
			set_variable = {
				name = tge_golden_eagle_progress_var
				value = tge_golden_eagle_progress_value
			}
			scope:journal_entry = {
				set_bar_progress = {
					name = tge_golden_eagle_progressbar
					value = owner.var:tge_golden_eagle_var
				}
			}
		}
	}

	on_monthly_pulse = {
		effect = {
			change_variable = {
				name = tge_golden_eagle_var
				add = var:tge_golden_eagle_progress_var
			}
			clamp_variable = {
				name = tge_golden_eagle_var
				max = 500
				min = 0
			}
			scope:journal_entry = {
				set_bar_progress = {
					name = tge_golden_eagle_progressbar
					value = owner.var:tge_golden_eagle_var
				}
			}
		}
	}
}
