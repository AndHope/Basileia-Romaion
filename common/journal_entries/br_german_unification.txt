	icon = "gfx/interface/icons/event_icons/waving_flag.dds"
	
	group = je_group_historical_content

	is_shown_in_lobby = {
		hidden_trigger = { NOT = { has_global_variable = schleswig_holstein_question_solved } }		
		OR = {
			country_has_primary_culture = cu:saxon
			country_has_primary_culture = cu:baltic_german
		}

		hidden_trigger = { NOT = { has_global_variable = schleswig_holstein_question_solved } }

		OR = {
			country_has_primary_culture = cu:saxon
			country_has_primary_culture = cu:baltic_german
		}			
	}
	
	status_desc = {
		first_valid = {
			triggered_desc = { 
				desc = je_schleswig_holstein_question_status
			}
		}
	}

    # Completion
    outcome = {
        header = je_standard_completion_header
        effect_header = je_standard_on_completion_header
        color = "gold_positive"
        conclusion_desc = je_standard_completion_conclusion_desc
        trigger = {
            trigger_if = {
                limit = {
                    OR = {
                        c:HAM ?= this
                        c:LUB ?= this
                        c:HOL ?= this
                        c:SCH ?= this
                    }
                }
                is_subject = no
            }
            trigger_if = {
                limit = {
                    NOT = {
                        OR = {
                            c:HAM ?= this
                            c:LUB ?= this
                            c:HOL ?= this
                            c:SCH ?= this
                        }
                    }
                    exists = c:DEN
                }
                custom_tooltip = {
                    text = je_schleswig_holstein_question_state_ownership_desc
                    any_scope_state = { state_region = s:STATE_SCHLESWIG_HOLSTEIN }
                    s:STATE_SCHLESWIG_HOLSTEIN = {
                        NOT = {
                            any_scope_state = {
                                OR = {
                                    NOT = {
                                        owner = {
                                            OR = {
                                                country_has_primary_culture = cu:saxon
                                                country_has_primary_culture = cu:baltic_german
                                            }
                                        }
                                    }
                                    owner = {
                                        is_subject_of = c:DEN
                                    }
                                }
                            }
                        }
                    }
                    s:STATE_ELBE = {
                        NOT = {
                            any_scope_state = {
                                OR = {
                                    NOT = {
                                        owner = {
                                            OR = {
                                                country_has_primary_culture = cu:saxon
                                                country_has_primary_culture = cu:baltic_german
                                            }
                                        }
                                    }
                                    owner = {
                                        is_subject_of = c:DEN
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        effect = {
            add_modifier = {
                name = br_modifier_schleswig_holstein_question_solver
                days = long_modifier_time
            }
            hidden_effect = { set_global_variable = schleswig_holstein_question_solved }
        }
        event_outcome_desc = {} # complex desc like a status_desc field
    }

    # Fail
    outcome = {
        header = je_standard_failure_header
        effect_header = je_standard_on_failure_header
        color = "blue_negative"
        conclusion_desc = je_standard_failure_conclusion_desc
        trigger = {
            custom_tooltip = {

                text = je_schleswig_holstein_question_solved_fail_trigger_desc
