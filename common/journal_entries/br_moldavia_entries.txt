
# Side with the Russians
je_side_russia = {

	group = je_group_foreign_affairs

	icon = "gfx/interface/icons/event_icons/event_default.dds"

	is_shown_in_lobby = {
		c:MOL ?= this
	}

    # Completion
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		conclusion_desc = je_side_russia_completed_desc
		color = "gold_positive"
		trigger = {
			is_at_war = no
			AND = {
				c:br_tartaria = {
					relations:ROOT >= relations_threshold:amicable
				}
				c:HUN = {
					relations:ROOT <= relations_threshold:cold
				}
			}
		}
		effect = {
			trigger_event = {
				id = moldavia_events.4
				popup = yes
			}

			add_journal_entry = {
				type = je_conquer_wallachia_dobruja
			}
		}
	}

	weight = 10000

	should_be_pinned_by_default_uninvolved_or_context = yes
}
